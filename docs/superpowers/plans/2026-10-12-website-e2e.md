# Website e2e on devnet Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `bun run e2e` drives the dashboard in a real browser against a dedicated test fleet on devnet and checks the
chain as well as the page, in four scenarios: read, kill switch, live steps, proposal verdicts.

**Architecture:** The fleet directory becomes `fleetDir()` (env `FLEET_DIR`, default `.devnet/fleet`), so a second
fleet, `.devnet/e2e-fleet/`, set up once by `bun scripts/setup.ts --e2e`, can be served by a Vite server on port 5174
that Playwright starts. Playwright (Node) reads the chain through a small Bun script, starts the Tick and Sentinel
agent processes per test, and restores the test fleet (resume, reject a waiting proposal) before and after every test.

**Tech Stack:** Bun, SvelteKit 3 / Vite 8, `@playwright/test` (Chromium), `pq-agent-guard` (the kit), `@solana/kit`.

**Spec:** `docs/superpowers/specs/2026-10-12-website-e2e-design.md`

## Review Focus

- **A run interrupted mid-test (Ctrl+C after Kill, or with a proposal waiting)** — the next run must still start from
  ACTIVE agents with no waiting proposal. Pinned by `e2e/restore.test.ts` (Task 3) and the auto fixture (Task 4).
- **A fresh worktree or clone whose `.env` names keypair files that do not exist** — the server then quietly shows an
  empty fleet; the run must stop before any test with the variable and path named. Pinned by
  `e2e/preflight-checks.test.ts` (Task 3).
- **The tests reaching the demo fleet by mistake** (same agent ids) — the page must show the `E2E …` names, and setup
  `--e2e` must never write to `.devnet/fleet/` or `.devnet/agents/`. Pinned by `profileFor` tests (Task 2),
  `fleetDir` tests (Task 1) and the read scenario's name check (Task 4).
- **An agent process that dies on start (bad env, port)** — the test must fail with its output, not time out silently.
  Pinned by the `running()` check and log attachment in the fixture (Task 4).
- **Browser console errors during a flow** (e.g. `Decision feed failed`) — any scenario must fail on them. Pinned by
  the `dashboard` fixture (Task 4).

`e2e/preflight-checks.ts`:

```ts
// What would make an e2e run fail for reasons outside the code under test, found before any test starts. Pure: the
// facts come from e2e/preflight.ts.

/** 0.01 SOL: a run is about a dozen devnet steps of ~15,000 lamports each, with room to spare. */
export const MIN_PAYER_LAMPORTS = 10_000_000n;

export interface KeypairFact {
	variable: string;
	path: string | undefined;
	exists: boolean;
}

export interface PreflightFacts {
	keypairs: KeypairFact[];
	/** Test-fleet agents without their on-chain identity file. */
	missingAgents: string[];
	/** null when the payer could not be read (its keypair problem is reported instead). */
	payerLamports: bigint | null;
}

export function preflightProblems({ keypairs, missingAgents, payerLamports }: PreflightFacts): string[] {
	const problems = keypairs.filter((keypair) => !keypair.exists).map(describeMissingKeypair);
	if (missingAgents.length > 0) {
		problems.push(`the test fleet lacks ${missingAgents.join(', ')}: run bun scripts/setup.ts --e2e`);
	}
	if (payerLamports !== null && payerLamports < MIN_PAYER_LAMPORTS) {
		problems.push(`the fee payer has ${payerLamports} lamports, below ${MIN_PAYER_LAMPORTS}: fund it on devnet`);
	}
	return problems;
}

function describeMissingKeypair({ variable, path }: KeypairFact): string {
	return path ? `${variable} in .env names ${path}, which does not exist` : `${variable} is not set in .env`;
}
```

`e2e/restore.ts`:

```ts
// Before and after every e2e test the test fleet goes back to its starting state: every agent active, no proposal
// waiting. A run stopped half-way (after a kill, or with a proposal open) is cleaned up by the next one. Pure: the
// fixture sends the requests.

export interface FleetCard {
	id: string;
	status: 'ACTIVE' | 'PAUSED' | 'UNAVAILABLE' | 'AWAITING_APPROVAL';
	currentTask: string;
	fleet: { pendingProposal: { counter: number } | null };
}

export interface RestoreRequest {
	path: string;
	body: Record<string, unknown>;
}

/** A killed agent shows PAUSED even with a proposal waiting, so it is resumed first. */
export function restoreRequests(cards: readonly FleetCard[]): RestoreRequest[] {
	return cards.flatMap((card) => [...resumeIfKilled(card), ...rejectIfWaiting(card)]);
}

/** Agents the server could not read, with its reason: no test can rely on them. */
export function unavailable(cards: readonly FleetCard[]): string[] {
	return cards.filter((card) => card.status === 'UNAVAILABLE').map((card) => `${card.id} (${card.currentTask})`);
}

function resumeIfKilled(card: FleetCard): RestoreRequest[] {
	return card.status === 'PAUSED' ? [{ path: `/api/agents/${card.id}/control`, body: { action: 'resume' } }] : [];
}

function rejectIfWaiting(card: FleetCard): RestoreRequest[] {
	const pending = card.fleet.pendingProposal;
	return pending
		? [{ path: `/api/agents/${card.id}/approval`, body: { verdict: 'reject', counter: pending.counter } }]
		: [];
}
```

```ts
function keypairFact(variable: string, path: string | undefined): KeypairFact {

async function payerLamports(rpcUrl: string, payer: KeypairFact): Promise<bigint | null> {

	const { address } = await loadKeypairSigner(payer.path);

		return loadConfig(process.env);
	} catch (error) {

	}
}

	missingAgents: E2E_PROFILE.agents.map((agent) => agent.id).filter((id) => !existsSync(onchainIdentityPath(id))),
	payerLamports: await payerLamports(config.rpcUrl, payer)
```

`e2e/agent-process.ts`:

```ts
// The test agents as their own processes, as in the demo, pointed at the test server. Stopped with SIGTERM so a step
// in flight lands first.
import { spawn } from 'node:child_process';
import { E2E_SENTINEL, E2E_TICK } from '../scripts/setup-plan';

export type AgentName = 'tick' | 'sentinel';

export interface AgentProcess {
	running(): boolean;
	output(): string;
	stop(): Promise<void>;
}

/** A devnet step can take up to ~30 s to land (scripts/demo.ts gives agents the same grace). */
const STOP_GRACE_MS = 40_000;
/** The server's step throttle allows one step per agent every 10 s; batched or not, a step lands about that often. */
const STEP_INTERVAL_MS = '10000';
/** The demo's Sentinel pace (scripts/demo.ts DEMO_PACE). */
const SENTINEL_SCHEDULE_MIN = '2';

function agentEnv(name: AgentName, apiUrl: string): Record<string, string> {
	return name === 'tick'
		? { TICK_API_URL: apiUrl, TICK_AGENT_ID: E2E_TICK.id, TICK_INTERVAL_MS: STEP_INTERVAL_MS }
		: {
				SENTINEL_API_URL: apiUrl,
				SENTINEL_AGENT_ID: E2E_SENTINEL.id,
				SENTINEL_INTERVAL_MS: STEP_INTERVAL_MS,
				SENTINEL_SCHEDULE_MIN
			};
}

export function startAgent(name: AgentName, apiUrl: string): AgentProcess {
	const child = spawn('bun', [`agents/${name}`], {
		env: { ...process.env, ...agentEnv(name, apiUrl) },
		stdio: ['ignore', 'pipe', 'pipe']
	});
	let output = '';
	const collect = (chunk: Buffer) => {
		output += chunk.toString();
	};
	child.stdout.on('data', collect);
	child.stderr.on('data', collect);
	const exited = new Promise<void>((resolve) => child.once('exit', () => resolve()));
	const running = () => child.exitCode === null && child.signalCode === null;
	return {
		running,
		output: () => output,
		async stop() {
			if (!running()) return;
			child.kill('SIGTERM');
			const forceStop = setTimeout(() => child.kill('SIGKILL'), STOP_GRACE_MS);
			await exited;
			clearTimeout(forceStop);
		}
	};
}
```

```ts
	{ button: 'Reject', chip: 'REJECTED' }
] as const;

test.use({ runningAgents: ['sentinel'] });

for (const { button, chip } of VERDICTS) {
	test(`records the owner's ${chip} verdict on a proposal on chain`, async ({ dashboard }) => {
		await dashboard.select(E2E_SENTINEL.name);
		await dashboard.armPriceShock();
		await expect(dashboard.card(E2E_SENTINEL.name)).toContainText('AWAITING_APPROVAL', {
			timeout: PROPOSAL_TIMEOUT_MS
		});
		const proposal = await dashboard.pendingProposalCounter();

		await dashboard.verdictButton(button).click();
		await expect(dashboard.card(E2E_SENTINEL.name)).toContainText('ACTIVE');
		await expect.poll(async () => (await readChain(E2E_SENTINEL.id)).counter, CHAIN_POLL).toBeGreaterThan(proposal);
		await expect(dashboard.feedEntry(proposal + 1)).toContainText(chip);
	});
}
```
