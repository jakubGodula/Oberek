> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A self-contained package `pq-agent-guard` in `kit/` (the program client, the guard, a one-agent API
`openGuardedAgent` / `step`), with the app importing it like any library.

**Tech Stack:** Bun (workspaces, `bun:test`), TypeScript, SvelteKit + Vite (dev server), svelte-check, Biome lint,
`@solana/kit` 8.4.0, `@noble/post-quantum`.

- [ ] **Step 1: Write the import-rewriting script**

`$SCRATCH/rewrite-imports.py` (not committed):

```python
#!/usr/bin/env python3
"""Rewrites the app's imports of the modules moving into kit/ to imports of pq-agent-guard.
Run from the worktree root BEFORE the git mv: it resolves import paths against the old layout."""
import os
import re
import subprocess

AGENT_DIR = 'src/lib/server/modules/agent'
GUARD_DIR = 'src/lib/server/modules/agent-guard'
MEMORY_GUARD = 'src/lib/server/modules/fleet/testing/memory-guard'
MOVING_AGENT_MODULES = {'instructions', 'message', 'transaction', 'guard-client', 'guard-errors', 'events', 'resume', 'files', 'idl'}
PACKAGE = 'pq-agent-guard'
TESTING = 'pq-agent-guard/testing'
MAX_LINE = 120
IMPORT = re.compile(r"^import\s+(type\s+)?\{([^}]*)\}\s+from\s+'([^']+)';\n", re.M)


def module_of(path):
    return re.sub(r'(\.test)?\.ts$', '', path)


def is_moving(path):
    module = module_of(path)
    directory, name = os.path.dirname(module), os.path.basename(module)
    return directory == GUARD_DIR or module == MEMORY_GUARD or (directory == AGENT_DIR and name in MOVING_AGENT_MODULES)


def package_for(importer, specifier):
    if not specifier.startswith('.'):
        return None
    target = os.path.normpath(os.path.join(os.path.dirname(importer), specifier))
    if target == MEMORY_GUARD:
        return TESTING
    if os.path.dirname(target) == GUARD_DIR:
        return PACKAGE
    if os.path.dirname(target) == AGENT_DIR and os.path.basename(target) in MOVING_AGENT_MODULES:
        return PACKAGE
    return None


def specifiers(type_only, body):
    names = [' '.join(raw.split()) for raw in body.split(',')]
    return [name if not type_only or name.startswith('type ') else f'type {name}' for name in names if name]


def render(names, package):
    ordered = sorted(set(names), key=lambda name: name.removeprefix('type ').lower())
    line = f"import {{ {', '.join(ordered)} }} from '{package}';"
    if len(line) <= MAX_LINE:
        return line + '\n'
    return 'import {\n' + ',\n'.join('\t' + name for name in ordered) + f"\n}} from '{package}';\n"


def rewrite(path):
    source = open(path).read()
    collected = {}

    def take(match):
        package = package_for(path, match.group(3))
        if package is None:
            return match.group(0)
        collected.setdefault(package, []).extend(specifiers(bool(match.group(1)), match.group(2)))
        return f'\0{package}\0'

    result = IMPORT.sub(take, source)
    for package, names in collected.items():
        result = result.replace(f'\0{package}\0', render(names, package), 1)
    result = re.sub(r'\0[^\0]*\0', '', result)
    if result != source:
        open(path, 'w').write(result)
        print('rewrote', path)


for path in subprocess.check_output(['git', 'ls-files', 'src', 'scripts', 'bot', 'agents'], text=True).split():
    if path.endswith(('.ts', '.svelte')) and not is_moving(path):
        rewrite(path)
```

```ts
import { createHash } from 'node:crypto';
import type { AgentGuard, GuardAccount } from './guard';

// One agent behind onchain_agent_guard: each action runs only after the program has recorded the decision behind it.

export interface GuardedStep<T> {
	/** The agent's counter after this step. */
	counter: number;
	/** Hex SHA-256 of the decision string, as recorded on chain. */
	decisionHash: string;
	/** What the action returned. */
	result: T;
}

export interface GuardedAgent {
	/**
	 * Records `decision` as the agent's next step, then runs `act`. A refused step throws GuardError and `act` does not
	 * run. If `act` throws, the step stays recorded: the record says the agent decided, not that the action succeeded.
	 */
	step<T>(decision: string, act: () => T | Promise<T>): Promise<GuardedStep<T>>;
	status(): Promise<GuardAccount>;
	/** Owner only: every further step is refused until resume. */
	kill(): Promise<GuardAccount>;
	/** Owner only: signed with the agent's ML-DSA-44 key; uses up one counter value. */
	resume(): Promise<GuardAccount>;
}

/** SHA-256 of the decision's UTF-8 bytes, hex: anyone holding the string can recompute what is on chain. */
export function hashDecision(decision: string): string {
	return createHash('sha256').update(Buffer.from(decision, 'utf8')).digest('hex');
}

export function createGuardedAgent(guard: AgentGuard, agentId: string): GuardedAgent {
	const read = () => guard.readAgent(agentId);
	return {
		async step<T>(decision: string, act: () => T | Promise<T>): Promise<GuardedStep<T>> {
			const decisionHash = hashDecision(decision);
			const { counter } = await guard.recordStep(agentId, (await read()).counter + 1, decisionHash);
			return { counter, decisionHash, result: await act() };
		},
		status: read,
		// AgentGuard takes the owner the caller claims; OnchainGuard checks its own owner key against the account.
		kill: async () => guard.kill(agentId, (await read()).owner),
		resume: async () => guard.resume(agentId, (await read()).owner)
	};
}
```

```ts
export const DEVNET_RPC_URL = 'https://api.devnet.solana.com';
/** OnchainGuard looks agents up by id; a guarded agent is one agent, so a fixed id names it. */
const GUARDED_AGENT_ID = 'guarded-agent';

export interface GuardedAgentOptions {
	/** The agent file `bun scripts/agent.ts register` wrote (AGENT_FILE): addresses and the ML-DSA-44 secret key. */
	agentFile: string;
	/** The owner's Solana keypair file: pays every fee and signs kill and resume. */

export async function openGuardedAgent(options: GuardedAgentOptions): Promise<GuardedAgent> {
	const rpcUrl = options.rpcUrl ?? DEVNET_RPC_URL;
	const identity = loadAgentFile(options.agentFile);
	const wallet = await loadKeypairSigner(options.wallet);

		owner: wallet,
		identityOf: () => identity
	});
	return createGuardedAgent(guard, GUARDED_AGENT_ID);
}
```

````markdown
# pq-agent-guard

Puts an existing agent behind the `onchain_agent_guard` program on Solana. Before each action, the agent's decision is

whatever the agent's code.

The kit talks to the program directly; you do not need the Oberek server. Oberek itself

`step` records the decision as the agent's next step and runs the action only after the step landed. The decision is a

| --- | --- |
| `AgentKilled` | the owner killed the agent |
| `AgentPending` | registered, not activated by the owner yet |
| `CounterMismatch` | another process recorded a step for this agent first; retry |
| `PayerUnfunded` | the wallet cannot pay the fee; nothing was sent |
| `AgentNotFound` | no such agent on this cluster |

## Kill and resume

`await agent.kill()` and `await agent.resume()`, or `bun scripts/agent.ts kill` / `resume` with the same `AGENT_FILE`.
Only the agent's owner can do either (`NotOwner` otherwise). A resume is signed with the agent's ML-DSA-44 key, so the
owner's Solana key alone cannot bring a killed agent back.

## What it protects, and what it does not

- **Protected:** the agent's decisions, as an on-chain record no one can forge without the ML-DSA-44 key, and the
  owner's kill switch, which the agent cannot get around through `step`.
- **Not protected:** the agent's own Solana wallet. It is still an Ed25519 key, as quantum-vulnerable as any wallet
  today.
- **Only what goes through `step`.** An action the agent runs without `step` is neither recorded nor stopped by a kill.
- **Ask before acting.** `step` stops an action only because it runs first. Recording after the action gives an audit
  trail but no kill switch.

## Gating your own on-chain program

An agent that skips `step` is not stopped. To stop it, keep the funds behind your own program and have that program
check the agent's account before it acts. Pass the `Agent` account in the instruction and check:

- it is the agent you expect (its address, kept in your program's config);
- its owner is the guard program `BL6Fjh4pADgjvZwaueeAcc7rYddFrdMNWnqK73x78ZSs`;
- its first 8 bytes are the `Agent` discriminator `[47, 166, 112, 147, 155, 197, 86, 7]`;
- byte 72 is `0` (Active). `1` is Killed, `2` is Pending.

A sketch in Anchor (not compiled in this repo; `kit/src/agent-layout.test.ts` pins the bytes above):

```rust
const GUARD_PROGRAM_ID: Pubkey = pubkey!("BL6Fjh4pADgjvZwaueeAcc7rYddFrdMNWnqK73x78ZSs");
const AGENT_DISCRIMINATOR: [u8; 8] = [47, 166, 112, 147, 155, 197, 86, 7];
const STATUS_OFFSET: usize = 72;
const ACTIVE: u8 = 0;

fn require_active_agent(agent: &AccountInfo, expected: &Pubkey) -> Result<()> {
    require_keys_eq!(*agent.key, *expected, GateError::WrongAgent);
    require_keys_eq!(*agent.owner, GUARD_PROGRAM_ID, GateError::WrongAgent);
    let data = agent.try_borrow_data()?;
    require!(data.len() > STATUS_OFFSET && data[..8] == AGENT_DISCRIMINATOR, GateError::WrongAgent);
    require!(data[STATUS_OFFSET] == ACTIVE, GateError::AgentNotActive);
    Ok(())
}
```

No CPI: the check reads the account. A kill takes effect for your program from the next transaction.
````
